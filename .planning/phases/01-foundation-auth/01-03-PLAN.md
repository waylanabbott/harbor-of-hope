GET  /api/auth/me              -> { isAuthenticated, email, roles[], supporterId? }
POST /api/auth/login           -> body: { email, password, twoFactorCode? } -> 200 or { requiresTwoFactor: true }
POST /api/auth/register        -> body: { email, password } -> 200
POST /api/auth/logout          -> 200
GET  /api/auth/providers       -> [{ displayName, name }]

GET  /api/auth/manage/2fa      -> { isMfaEnabled, hasAuthenticator, recoveryCodesLeft }
POST /api/auth/manage/2fa/setup -> { sharedKey, authenticatorUri }
POST /api/auth/manage/2fa/verify -> body: { code } -> { recoveryCodes[] }
POST /api/auth/manage/2fa/disable -> 200
