   - `import { Tooltip as MuiTooltip } from '@mui/material';` (if not already imported -- check for conflict with existing Tooltip import)

2. Add state:
   - `const [churnPredictions, setChurnPredictions] = useState<Map<number, ChurnPrediction>>(new Map());`

3. Add a second useEffect that fires when `supporters` changes:
   ```typescript
   useEffect(() => {
     if (!supporters?.items?.length) return;
     const ids = supporters.items.map(s => s.supporterId);
     fetchBatchChurnPredictions(ids)
       .then(predictions => {
         const map = new Map(predictions.map(p => [p.supporterId, p]));
         setChurnPredictions(map);
       })
       .catch(() => {
         // Silently fail -- churn badges are non-critical enhancement
       });
   }, [supporters]);
   ```

4. Add a new column to the `columns` array, inserting it AFTER the 'status' column and BEFORE the 'donationCount' column:
   ```typescript
   {
     id: 'churnRisk' as any,
     label: 'Churn Risk',
     minWidth: 100,
     align: 'center' as const,
     render: (row: SupporterItem) => {
       const prediction = churnPredictions.get(row.supporterId);
       if (!prediction) return <Chip label="..." size="small" variant="outlined" />;
       return (
         <MuiTooltip title={`${(prediction.churnProbability * 100).toFixed(0)}% churn probability`}>
           <span>
             <RiskBadge level={prediction.riskLevel} />
           </span>
         </MuiTooltip>
       );
     },
   }
   ```
   Note: The `as any` cast on `id` is needed because 'churnRisk' is not a key of SupporterItem -- it's a virtual/computed column. The render function handles the display.

   IMPORTANT: Since `columns` is currently defined as a const outside the component (line 44), and it needs access to `churnPredictions` state, you must move the columns definition INSIDE the component function body and wrap it in `useMemo` depending on `churnPredictions`. Or alternatively, keep columns outside and add the churn column dynamically using a computed `allColumns` inside the component. Prefer the latter approach for minimal diff:
   ```typescript
   // Inside the component, after churnPredictions state:
   const allColumns: Column<SupporterItem>[] = [
     ...columns.slice(0, 5), // up to and including 'status'
     {
       id: 'churnRisk' as any,
       label: 'Churn Risk',
       // ... render function as above using churnPredictions from closure
     },
     ...columns.slice(5), // 'donationCount' and beyond
   ];
   ```
   Then pass `allColumns` to DataTable instead of `columns`.
